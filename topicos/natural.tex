\section{Amostragem Natural}

\begin{frame}{Amostragem Natural}{Contexto}
	
	A amostragem ideal utiliza um trem de impulsos de Dirac $\delta(t)$, que é uma abstração matemática puramente teórica e fisicamente inexequível, pois exige pulsos com largura infinitesimal ($\tau \to 0$) e amplitude infinita para reter energia finita. 
	
	A amostragem natural surge para tornar o processo fisicamente realizável: em vez de impulsos instantâneos, utiliza-se um trem de pulsos retangulares de largura finita $\tau$ e período $T_s$, gerados por chaves eletrônicas reais (como transistores operando em comutação) que se fecham por um intervalo $\tau$ a cada ciclo. Ela aproxima o sistema de um cenário real de chaveamento analógico.
	
\end{frame}

% Definição do Pulso p0(t)
\begin{frame}{Amostragem Natural}{Definição do Pulso Fundamental $p_0(t)$}
	
	O processo de amostragem natural utiliza pulsos retangulares de largura finita $\tau$. Definimos inicialmente o pulso unitário $p_0(t)$ centrado na origem:
	
	\begin{equation}
		p_0(t) = 
		\begin{cases}
			1, & -\frac{\tau}{2} \le t \le \frac{\tau}{2} \\[6pt]
			0, & \text{caso contrário}
		\end{cases}
		\label{eq: definicao_pulso_unitario_natural}
	\end{equation}

% Conexão e Série de Fourier de p(t)
\begin{frame}{Amostragem Natural}{Expansão em Série de Fourier de $p(t)$}
	
	Como $p(t)$ é estritamente periódico com período $T_s$ ($f_s = 1/T_s$), podemos representá-lo via Série Exponencial de Fourier:
	
	\begin{equation}
		p(t) = \sum_{k=-\infty}^{\infty} P[k] e^{j 2\pi k f_s t}
		\label{eq: representacao_serie_fourier}
	\end{equation}
	
	O cálculo dos coeficientes $P[k]$ é realizado no intervalo de um período $[-\frac{T_s}{2}, \frac{T_s}{2}]$:
	
	\begin{equation}
		P[k] = \frac{1}{T_s} \int_{-T_s/2}^{T_s/2} p(t) e^{-j 2\pi k f_s t} \, dt
		\label{eq: representacao_coeficiente_fourier_natural} 
	\end{equation}
	
\end{frame}
	
\begin{frame}{Amostragem Natural}{Expansão em Série de Fourier de $p(t)$}
	
	Dentro deste intervalo $[-T_s/2, T_s/2]$, o sinal periódico $p(t)$ é igual ao pulso fundamental $p_0(t)$ (com $\tau < T_s$). Tomando o sinal $p_0(t)$ no intervalo $[-\tau/2, \tau/2]$:
	
	\begin{align*}
		P[k] &= \frac{1}{T_s} \int_{-T_s/2}^{T_s/2} p_0(t) e^{-j 2\pi k f_s t} \, dt = \frac{1}{T_s} \int_{-\tau/2}^{\tau/2} (1) e^{-j 2\pi k f_s t} \, dt \\[6pt]
		&= \frac{1}{T_s} \left[ \frac{e^{-j 2\pi k f_s t}}{-j 2\pi k f_s} \right]_{-\tau/2}^{\tau/2} = \frac{1}{T_s} \left( \frac{e^{j \pi k f_s \tau} - e^{-j \pi k f_s \tau}}{j 2\pi k f_s} \right)
	\end{align*}
	
	Aplicando a identidade de Euler $\sin(\theta) = \frac{e^{j\theta} - e^{-j\theta}}{2j}$:
	\begin{equation*}
		P[k] = \frac{\sin(\pi k f_s \tau)}{\pi k T_s f_s}
	\end{equation*}
	
\end{frame}

\begin{frame}{Amostragem Natural}{Coeficientes $P[k]$ da Série de Fourier}
		
	Como $T_s f_s = 1$, multiplicamos e dividimos por $\tau$ para obter a função $\text{sinc}$:
	\begin{equation}
		P[k] = \frac{\tau}{T_s} \cdot \frac{\sin(\pi k f_s \tau)}{\pi k f_s \tau} \implies P[k] = \frac{\tau}{T_s} \text{sinc}(k f_s \tau)
		\label{eq: definicao_pk_natural}
	\end{equation}
	
	\begin{block}{Sinal de Amostragem no Tempo}
		\begin{equation}
			p(t) = \sum_{k=-\infty}^{\infty} \left[ \frac{\tau}{T_s} \text{sinc}(k f_s \tau) \right] e^{j 2\pi k f_s t}
			\label{eq: sinal_amostragem_tempo_natural}
		\end{equation}
	\end{block}
\end{frame}

% Construção Analítica de s(t) no Tempo
\begin{frame}{Amostragem Natural}{Modelagem Analítica no Tempo}
	O sinal amostrado $s(t)$ é gerado pelo produto direto entre $x(t)$ e $p(t)$:
	
	\begin{equation}
		s(t) = x(t) \cdot p(t)
		\label{eq: definicao_amostragem_natural}
	\end{equation}
	
	Substituindo a expansão em Série de Fourier de $p(t)$:
	
	\begin{equation*}
		s(t) = x(t) \cdot \sum_{k=-\infty}^{\infty} P[k] e^{j 2\pi k f_s t}
	\end{equation*}
	
	
\end{frame}

\begin{frame}{Amostragem Natural}{Modelagem Analítica no Tempo}
	Pela propriedade distributiva, inserimos $x(t)$ no somatório:
	
	\begin{equation*}
		s(t) = \sum_{k=-\infty}^{\infty} P[k] x(t) e^{j 2\pi k f_s t}
	\end{equation*}
	
	
	Substituindo a expressão analítica de $P[k]$ (\ref{eq: definicao_pk}):
	
	\begin{equation*}
		s(t) = \sum_{k=-\infty}^{\infty} \left[ \frac{\tau}{T_s} \text{sinc}(k f_s \tau) \right] x(t) e^{j 2\pi k f_s t}
	\end{equation*}
\end{frame}

% analise no dominio da frequencia
\begin{frame}{Amostragem Natural}{Análise no Domínio da Frequência}
	
	A multiplicação no tempo equivale à convolução no domínio da frequência  \cite{oppenheim1997signals}:
	\begin{equation}
		S(f) = \mathcal{F}\{x(t) \cdot p(t)\} = X(f) * P(f)
		\label{eq: definicao_convolucao_frequencia_natural}
	\end{equation}
	
	Sabemos que a expansão em Série de Fourier de $p(t)$ é: 
	\begin{equation*}
		p(t) = \sum_{k=-\infty}^{\infty} P[k] e^{j k \omega_s t}
	\end{equation*}
	
	Na frequência angular $\omega$ (rad/s), a Transformada de Fourier de um sinal periódico gera impulsos multiplicados por $2\pi$  \cite{oppenheim1997signals}:
	\begin{equation*}
		P(j\omega) = 2\pi \sum_{k=-\infty}^{\infty} P[k] \delta(\omega - k\omega_s)
	\end{equation*}
	
\end{frame}

\begin{frame}{Amostragem Natural}{Análise no Domínio da Frequência}
	
	Para converter a transformada para Hertz ($f$), utilizamos as relações $\omega = 2\pi f$ e $\omega_s = 2\pi f_s$:
	\begin{equation*}
		P(f) = 2\pi \sum_{k=-\infty}^{\infty} P[k] \delta(2\pi f - 2\pi k f_s)
	\end{equation*}
	
	Fatorando a constante $2\pi$ dentro da função impulso:
	\begin{equation*}
		P(f) = 2\pi \sum_{k=-\infty}^{\infty} P[k] \delta\big(2\pi(f - k f_s)\big)
	\end{equation*}
	
	Aplicando a propriedade de escalonamento do impulso de Dirac, $\delta(ax) = \frac{1}{|a|}\delta(x)$  \cite{oppenheim1997signals}:
	\begin{equation*}
		P(f) = 2\pi \sum_{k=-\infty}^{\infty} P[k] \left[ \frac{1}{2\pi} \delta(f - k f_s) \right]
	\end{equation*}
	
\end{frame}

\begin{frame}{Amostragem Natural}{Análise no Domínio da Frequência}
	Os fatores $2\pi$ se cancelam, resultando na equação final do espectro de amostragem:
	\begin{equation}
		P(f) = \sum_{k=-\infty}^{\infty} P[k] \delta(f - k f_s)
		\label{eq: pf_natural}
	\end{equation}
	
	Calculando a convolução $X(f) * P(f)$:
	\begin{align*}
		S(f) &= X(f) * \sum_{k=-\infty}^{\infty} P[k] \delta(f - k f_s) \\[4pt]
		&= \sum_{k=-\infty}^{\infty} P[k] \left[ X(f) * \delta(f - k f_s) \right] \\[4pt]
		&= \sum_{k=-\infty}^{\infty} P[k] X(f - k f_s)
	\end{align*}
\end{frame}

		H_r(f)=

	\end{equation}

% Sinal Reconstruído no Tempo
\begin{frame}{Reconstrução da Amostragem Natural}{Sinal Reconstruído no Tempo}
	
	Aplicando a Transformada Inversa de Fourier ao espectro recuperado, $y(t) = \mathcal{F}^{-1}\{Y(f)\}$, obtemos exatamente o sinal original no tempo, sem distorções.
	
	\vspace{0.5em}
	
	\def\Ts{0.2}
	\def\Tau{0.05}
	\def\Nam{5}
	
	\begin{figure}
		\centering
		\begin{tikzpicture}
			\begin{axis}[
				pamaxis,
				width=0.9\textwidth,
				height=4.0cm,
				ylabel={$y(t) \approx x(t)$},
				xlabel={$t$},
				trig format=rad,
				ymin=-1.3, ymax=1.45,
				xmin=-0.1, xmax=1.1,
				axis lines=middle
				]
				
				% Trem de pulsos atenuado no fundo (Para dar o contexto de onde ele veio)
				\pgfplotsextra{%
					\foreach \n in {0,...,\Nam}{%
						\pgfmathsetmacro\xa{\n*\Ts - \Tau/2}%
						\pgfmathsetmacro\xb{\n*\Ts + \Tau/2}%
						
						\draw[
						gray!40,
						line width=.6pt,
						fill=gray!10
						]
						(axis cs:\xa,0) --
						(axis cs:\xa,{cos(4*pi*\xa)}) --
						plot[
						domain=\xa:\xb,
						samples=25,
						smooth
						]
						(axis cs:\x,{cos(4*pi*\x)}) --
						(axis cs:\xb,{cos(4*pi*\xb)}) --
						(axis cs:\xb,0) --
						cycle;
					}%
				}
				
				% Sinal x(t) reconstruído perfeitamente
				\addplot[
				sinalcor,
				line width=1.5pt,
				smooth,
				domain=0:1,
				samples=300
				]
				{cos(4*pi*x)};
				
				
			\end{axis}
		\end{tikzpicture}
		\caption{Sinal contínuo recuperado a partir dos pulsos de amostragem natural.}
		\label{fig: sinal_natural_reconstruido}
	\end{figure}
	
	\vspace{0.2em}
	{\small Se o Critério de Nyquist ($f_s > 2f_m$) for satisfeito, a informação contida nos topos dos pulsos é suficiente para interpolar e restaurar a onda contínua $x(t)$.}
	
\end{frame}
