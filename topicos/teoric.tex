\section{Teorema de Amostragem}

% =========================================================
% AMOSTRAGEM
% =========================================================

\begin{frame}{Teorema de Amostragem}{Definição}

	Representar um sinal contínuo $x(t)$ pelos seus valores em instantes
	discretos \cite{oppenheim1997signals}:

	\vspace{0.8em}
	\begin{columns}[T]
		\begin{column}{0.33\textwidth}
			\centering
			\textbf{Instantes}
			\[ t = nT_s,\quad n\in\mathbb{Z} \]
		\end{column}
		\begin{column}{0.33\textwidth}
			\centering
			\textbf{Taxa}
			\[ f_s = \frac{1}{T_s} \]
		\end{column}
		\begin{column}{0.33\textwidth}
			\centering
			\textbf{Amostras}
			\[ x[n] = x(nT_s) \]
		\end{column}
	\end{columns}

	\vspace{1.2em}
	\begin{itemize}
		\item $T_s$: período de amostragem; $f_s$: frequência de amostragem.
		\item É o que viabiliza o processamento digital de sinais contínuos.
	\end{itemize}

\end{frame}


% =========================================================
% NYQUIST-SHANNON
% =========================================================

\begin{frame}{Teorema de Amostragem}{Nyquist-Shannon}

	Sinal limitado em banda, com maior frequência $f_{\max}$: a reconstrução
	perfeita exige \cite{oppenheim1997signals}

	\[
	\boxed{f_s>2f_{\max}}
	\qquad\qquad
	2f_{\max} = \text{taxa de Nyquist}
	\]

	\vspace{0.8em}
	\begin{itemize}
		\item $f_s>2f_{\max}$ $\Rightarrow$ réplicas espectrais não se sobrepõem;
		\item réplicas separadas $\Rightarrow$ um passa-baixas isola a banda base;
		\item maior a banda do sinal, maior a taxa necessária.
	\end{itemize}

\end{frame}


% =========================================================
% ALIASING E ANTI-ALIASING
% =========================================================

\begin{frame}{Teorema de Amostragem}{Aliasing e Filtro Anti-Aliasing}

	\textbf{Aliasing:} taxa insuficiente ($f_s\le2f_{\max}$) $\Rightarrow$ as
	réplicas se sobrepõem e uma componente alta é lida como uma componente
	de frequência menor \cite{oppenheim1997signals}.

	\vspace{0.8em}
	\textbf{Filtro anti-aliasing (AA):} atua \textbf{antes} da amostragem.
	\begin{itemize}
		\item Limita a banda do sinal, fixando $f_{\max}$;
		\item atenua tudo acima do que a taxa consegue representar.
	\end{itemize}

\end{frame}


% =========================================================
% MÉTODOS DE AMOSTRAGEM
% =========================================================

\begin{frame}{Teorema de Amostragem}{Métodos de Amostragem}

	Os métodos diferem na \textbf{forma como cada amostra é mantida no tempo}
	\cite{oppenheim1997signals,lathi2012sistemas}:

	\vspace{0.6em}
	\begin{itemize}
		\item \textbf{Ideal:} impulsos de Dirac --- a amostra existe só no
		      instante $t=nT_s$;
		      \vspace{-0.3em}
		      \[
		      x_s(t)=\sum_{n=-\infty}^{\infty} x(nT_s)\,\delta(t-nT_s)
		      \]
		\item \textbf{Natural:} pulso de largura $\tau$ que \textbf{acompanha}
		      o sinal durante o pulso \cite{lathi2012sistemas};
		\item \textbf{Flat-top} (\textit{sample-and-hold}): pulso de largura
		      $\tau$ com valor \textbf{constante} --- patamares.
	\end{itemize}

	\vspace{0.8em}
	Adiante: os três métodos, o efeito da largura $\tau$ e a respectiva
	reconstrução.

\end{frame}
