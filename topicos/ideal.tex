\begin{frame}{Método Ideal: Amostragem}{Domínio do Tempo}
\small
Pela propriedade de \textbf{peneiramento},
$x_f(t)\,\delta(t-nT_s) = x_f(nT_s)\,\delta(t-nT_s)$:
\begin{equation*}
  x_\delta(t) = x_f(t)\cdot \idamos{p(t)}
  = \boxed{\;\sum_{n=-\infty}^{\infty} x_f(nT_s)\, \delta(t - nT_s)\;},
  \qquad
  \idamos{p(t) = \sum_{n=-\infty}^{\infty} \delta(t - nT_s)}
\end{equation*}
\begin{center}
\begin{tikzpicture}
\begin{groupplot}[group style={group name=g, group size=3 by 1, horizontal sep=1.3cm}, ftdemot, height=0.18\textheight]
\nextgroupplot[title={(a) Sinal original\\[1pt]{\tiny\ftstrut $x_f(t)$}}]
  \addplot[ftazul!40!black, thick, domain=0:\fttmax, samples=300] {cos(deg(2*pi*\ftfzero*x))};

  \addplot[ftimppos, ftazul] coordinates {\ftimpt};
  \draw[gray, <->, >=stealth] (axis cs:\ftTs,1.18) -- node[above, font=\tiny, inner sep=1pt]{$T_s$} (axis cs:2*\ftTs,1.18);
\nextgroupplot[title={(c) Sinal amostrado\\[1pt]{\tiny\ftstrut $x_\delta(t)$}}]
  \addplot[gray!55, dashed, domain=0:\fttmax, samples=300] {cos(deg(2*pi*\ftfzero*x))};
  \addplot[ycomb, ftazul, thick] coordinates {\ftamostras};

\vspace{-0.3em}
Sem pulso de retenção: o sinal amostrado só existe em $t = nT_s$ --- entre as
amostras, nada.
\end{frame}

\begin{frame}{Método Ideal: Amostragem}{Domínio da Frequência}
\small
Produto no tempo $\Rightarrow$ convolução na frequência: cada impulso de $\idamos{P(f)}$ replica $X_f(f)$.
\begin{equation*}
  X_\delta(f) = X_f(f) * \idamos{P(f)},
  \quad
  \idamos{P(f) = \frac{1}{T_s}\sum_{n=-\infty}^{\infty} \delta(f - nf_s)}
  \;\Longrightarrow\;
  \boxed{\; X_\delta(f) = f_s\sum_{n=-\infty}^{\infty} X_f(f - nf_s) \;}
\end{equation*}
\vspace{-1.2em}
\begin{center}

\nextgroupplot[title={(b) Trem de impulsos\\[1pt]{\tiny\ftstrut $P(f)$}}, yticklabels={$\frac{1}{T_s}$}]
  \addplot[ftimpulso, ftazul] coordinates {\ftimpf};
\nextgroupplot[title={(c) Sinal amostrado\\[1pt]{\tiny\ftstrut $X_\delta(f)$}}, yticklabels={$\frac{1}{2T_s}$}]
  \addplot[ftimpulso, ftazul] coordinates {\ftrepideal};
\end{groupplot}
\ftop{1}{2}{*}
\ftop{2}{3}{=}
\end{tikzpicture}
\end{center}
\vspace{-0.6em}
\begin{itemize}\setlength{\itemsep}{0pt}
  \item \textbf{Réplicas de amplitude constante}: nenhum envelope pondera o espectro.
  \item Réplica vizinha em $f_s - f_{max}$: $f_s - f_{max} > f_{max} \Rightarrow f_s > 2f_{max}$.
\end{itemize}
\end{frame}

% ----------------------------------------------------------------------------
%  B. APLICAÇÃO AOS PARÂMETROS FIXOS
% ----------------------------------------------------------------------------
\begin{frame}{Método Ideal: Aplicação}{Parâmetros Fixos --- Domínio do Tempo}
    \small
    $x_f(t) = \cos(2\pi \cdot 2t)$, \; $f_s = 5$~Hz, \; $T_s = 0,2$~s:
    \begin{align*}
        x_\delta(t) &= \sum_{n=-\infty}^{\infty} x_f(nT_s)\, \delta(t - nT_s)
        = \sum_{n=-\infty}^{\infty} \cos\!\left(2\pi \cdot 2 \cdot \frac{n}{5}\right) \delta\!\left(t - \frac{n}{5}\right)\\
        &= \sum_{n=-\infty}^{\infty} \cos\!\left(\frac{4\pi n}{5}\right) \delta\!\left(t - \frac{n}{5}\right)
    \end{align*}

    \[
        \begin{array}{c|ccccc}
            n & 0 & 1 & 2 & 3 & 4 \\ \hline
            x_f(nT_s) & 1 & -0{,}809 & 0{,}309 & 0{,}309 & -0{,}809
        \end{array}
    \]

    \begin{itemize}
      \item Padrão se repete a cada 5 amostras, pois $f_0/f_s = 2/5$.
      \item $2{,}5$ amostras por período: o mínimo exigido por $f_s > 2f_{max}$.
    \end{itemize}
\end{frame}

\begin{frame}{Método Ideal: Aplicação}{Parâmetros Fixos --- Domínio da Frequência}
    \small
    $$X_\delta(f) = f_s \sum_{k=-\infty}^{\infty} X_f(f - k f_s) = 5 \sum_{k=-\infty}^{\infty} X_f(f - 5k)$$

    $$
    \left.\begin{matrix}
    k=0: \pm 2\text{ Hz} \\
    k=1: 3 \text{ e } 7\text{ Hz} \\
    k=-1: -7 \text{ e } -3\text{ Hz} \\
    k=2: 8 \text{ e } 12\text{ Hz}
    \end{matrix}\right\} \text{componentes em } \pm 2, \pm 3, \pm 7, \pm 8, \pm 12, \dots
    $$

    \begin{itemize}
      \item Peso de cada réplica: $f_s \cdot \tfrac{1}{2} = 2{,}5$ --- \textbf{todas iguais}.
      \item Banda base ($2$~Hz) e primeira réplica ($3$~Hz) separadas: sem
            \textit{aliasing}.
    \end{itemize}
\end{frame}

% ----------------------------------------------------------------------------
%  C. RECONSTRUÇÃO
% ----------------------------------------------------------------------------
\begin{frame}{Método Ideal: Reconstrução}{Filtro Passa-Baixas Ideal}
    \small
    Réplicas separadas $\Rightarrow$ basta descartar todas menos a banda base:
    passa-baixas ideal de corte $f_s/2$ e ganho $T_s$.
    \[
        H_{PB}(f) =
        \begin{cases}
            T_s, & |f| < \dfrac{f_s}{2}\\[4pt]
            0, & \text{caso contrário}
        \end{cases}
        \qquad\longrightarrow\qquad
        \begin{cases}
            0{,}2, & |f| < 2{,}5\,\text{Hz}\\
            0, & \text{caso contrário}
        \end{cases}
    \]

    \[
        \underbrace{2\,\text{Hz}}_{\text{banda base}} < \underbrace{2{,}5\,\text{Hz}}_{\text{corte}} < \underbrace{3\,\text{Hz}}_{\text{1ª réplica}}
        \qquad \Longrightarrow \qquad
        \text{margem de } 0{,}5\,\text{Hz de cada lado}
    \]

    \begin{itemize}
      \item Banda de guarda de $1$~Hz, obtida com $f_s = 5$~Hz.
      \item Com $f_s \le 4$~Hz, a réplica em $f_s - 2$ invadiria a banda base.
    \end{itemize}
\end{frame}

\begin{frame}{Método Ideal: Reconstrução}{Domínio da Frequência}
\small
Produto do espectro amostrado pelo passa-baixas: o ganho $T_s$ cancela o fator
$f_s$ da amostragem.
\begin{equation*}
  X_r(f) = X_\delta(f)\cdot \idrec{H_{PB}(f)} = T_s\cdot\frac{1}{T_s}X_f(f) = \boxed{\,X_f(f)\,}
  \qquad
  \text{(} 0{,}2 \cdot 5 \cdot \tfrac{1}{2} = \tfrac{1}{2}\text{)}
\end{equation*}
\begin{center}
\begin{tikzpicture}
\begin{groupplot}[group style={group name=g, group size=3 by 1, horizontal sep=1.3cm}, ftdemof, height=0.18\textheight]

Sem ponderação por $\mathrm{sinc}$ e sem termo de fase: espectro \textbf{idêntico}
ao original --- nenhum equalizador.
\pause\pause
\end{frame}

\begin{frame}{Método Ideal: Reconstrução}{Domínio do Tempo}
\small
Convolução com $h_{PB}(t) = \mathrm{sinc}(f_s t)$: cada impulso vira um $\mathrm{sinc}$ ponderado pela sua amostra.
\begin{equation*}
  x_r(t) = x_\delta(t) * \idrec{h_{PB}(t)}
  = \sum_{n=-\infty}^{\infty} x_f(nT_s)\, \idrec{\mathrm{sinc}\!\big(f_s(t - nT_s)\big)}
  = \boxed{\, x_f(t) \,}
\end{equation*}
