
\section{Simulação}

\begin{frame}[allowframebreaks]{Simulação}{Parâmetros fixos}
    \small

    \textbf{Sinal original e espectro:}
    \begin{align*}
        x(t) &= \cos(2\pi \cdot 2t) + 0,5\cos(2\pi \cdot 18t) \\
        &\Updownarrow \\
        X(f) &= \frac{1}{2}[\delta(f-2) + \delta(f+2)]
        + \frac{0,5}{2}[\delta(f-18) + \delta(f+18)]
    \end{align*}

    \textbf{Filtro Anti-aliasing ($H_{AA}$):}
    \begin{align*}
        X_f(f) &= X(f) \cdot H_{AA}(f)
        \longrightarrow
        H_{AA}(f) =
        \begin{cases}
            1, & |f| \le 8\,\text{Hz}\\
            0, & |f| > 8\,\text{Hz}
        \end{cases}
        \quad \leftarrow f_c \\[2pt]
        X_f(f) &= \frac{1}{2}[\delta(f-2) + \delta(f+2)]
        \longleftrightarrow
        x_f(t) = \cos(2\pi \cdot 2t)
    \end{align*}

    \textbf{Parâmetros de Amostragem:}
    \[
        f_{max} = 2\,\text{Hz}
        \implies
        f_s > 2f_{max} = 4\,\text{Hz}
    \]

    A partir dessa preparação, utiliza-se a equação filtrada em banda para realizar as amostragens: \begin{align*} x_f(t) = \cos(2\pi \cdot 2t) \end{align*}

\end{frame}

\begin{frame}{Caso de Teste 1}{Cenário Ideal}
	\small
	
	\textbf{Parâmetros da simulação:}
	
	\begin{center}
		\begin{tabular}{ll}
			\hline
			Sinal de entrada & $x(t)=\cos(2\pi\cdot2t)+0.5\cos(2\pi\cdot18t)$ \\
			Corte do anti-aliasing & $f_{c,aa}=4$~Hz \\
			Frequência de amostragem & $f_s=6$~Hz \\
			Período de amostragem & $T_s=1/f_s=1/6\approx0.1667$~s \\
			Largura dos pulsos & $\tau=0.1$~s \\
			Corte da reconstrução & $f_{c,rec}=3$~Hz \\
			\hline
		\end{tabular}
	\end{center}
	
	\vspace{0.5em}
	
	\textbf{Comportamento esperado:}
	
	\begin{itemize}
		\item \textbf{Filtro AA:} elimina a componente de 18~Hz, restando apenas a senoide de 2~Hz.
		\item \textbf{Amostragem:} $f_s=6$~Hz respeita Nyquist para $f_{\max}=2$~Hz, pois
		$6>2\cdot2=4$~Hz.
		\item \textbf{Reconstrução:} o filtro com corte em $3$~Hz isola a banda base e elimina as réplicas espectrais.
		\item O sinal reconstruído esperado é uma senoide de 2~Hz, correspondente ao sinal filtrado.
	\end{itemize}
	
\end{frame}

\begin{frame}{Caso de Teste 2}{Aliasing Clássico — Falha do Filtro Anti-Aliasing}
	\small
	
	\textbf{Parâmetros da simulação:}
	
	\begin{center}
		\begin{tabular}{ll}
			\hline
			Sinal de entrada & $x(t)=\cos(2\pi\cdot2t)+0.5\cos(2\pi\cdot18t)$ \\
			Corte do anti-aliasing & $f_{c,aa}=20$~Hz \\
			Frequência de amostragem & $f_s=5$~Hz \\
			Período de amostragem & $T_s=1/f_s=0.2$~s \\
			Largura dos pulsos & $\tau=0.05$~s \\
			Corte da reconstrução & $f_{c,rec}=2.5$~Hz \\
			\hline
		\end{tabular}
	\end{center}
	
	\vspace{0.5em}
	
	\textbf{Comportamento esperado:}
	
	\begin{itemize}
		\item \textbf{Filtro AA:} com $f_{c,aa}=20$~Hz, a componente de 18~Hz permanece no sinal.
		\item \textbf{Amostragem:} $f_s=5$~Hz é insuficiente para representar a componente de 18~Hz.
		\item A componente sofre \textbf{aliasing}, produzindo réplicas em frequências da forma $f_{\text{réplica}}=|f_{\text{original}}-k f_s|.$
		\item \textbf{Reconstrução:} as componentes resultantes interferem na banda base, produzindo um sinal temporal distorcido.
	\end{itemize}
	
\end{frame}

\begin{frame}{Caso de Teste 3}{Aliasing da Frequência Fundamental — Violação de Nyquist}
	\small
	
	\textbf{Parâmetros da simulação:}
	
	\begin{center}
		\begin{tabular}{ll}
			\hline
			Sinal de entrada & $x(t)=\cos(2\pi\cdot2t)+0.5\cos(2\pi\cdot18t)$ \\
			Corte do anti-aliasing & $f_{c,aa}=10$~Hz \\
			Frequência de amostragem & $f_s=3$~Hz \\
			Período de amostragem & $T_s=1/f_s\approx0.3333$~s \\
			Largura dos pulsos & $\tau=0.1$~s \\
			Corte da reconstrução & $f_{c,rec}=1.5$~Hz \\
			\hline
		\end{tabular}
	\end{center}
	
	\vspace{0.5em}
	
	\textbf{Comportamento esperado:}
	
	\begin{itemize}
		\item \textbf{Filtro AA:} elimina a componente de 18~Hz, preservando a componente de 2~Hz.
		\item \textbf{Amostragem:} $f_s=3$~Hz viola o critério de Nyquist, pois
		$f_s=3 < 2f_{\max}=4~\text{Hz}$.
		\item A componente de 2~Hz sofre aliasing e é rebatida para $|2-3|=1~\text{Hz}$.
		\item \textbf{Reconstrução:} o filtro com $f_{c,rec}=1.5$~Hz seleciona a componente de 1~Hz.
		\item O sinal reconstruído será uma senoide limpa de 1~Hz, embora o sinal original filtrado possua 2~Hz.
	\end{itemize}
	
\end{frame}
